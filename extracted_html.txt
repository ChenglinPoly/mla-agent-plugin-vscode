    // 简单 Markdown 渲染（必须在 appendMessage 之前定义）
    function renderMarkdown(text) {
      if (!text) {
        console.log('[renderMarkdown] 文本为空');
        return '';
      }
      
      try {
        console.log('[renderMarkdown] 输入:', text.substring(0, 50));
        
        let html = text;
        
        // 转义 HTML
        const tempDiv = document.createElement('div');
        tempDiv.textContent = html;
        html = tempDiv.innerHTML;
        
        console.log('[renderMarkdown] 转义后:', html.substring(0, 50));
        
        // 粗体
        html = html.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>');
        
        // 行内代码（使用 \\x60 代替反引号，避免与外层模板字符串冲突）
        html = html.replace(/\\x60([^\\x60]+)\\x60/g, '<code>$1</code>');
        
        // 换行（\n 不是 \\n）
        html = html.replace(/\n/g, '<br>');
        
        console.log('[renderMarkdown] 最终:', html.substring(0, 50));
        
        return html;
      } catch (e) {
        console.error('[renderMarkdown] 错误:', e, '输入:', text);
        return escapeHtml(text);
      }
    }

    function updateAssistantMessage(id, content) {

          // 隐藏 XML 标签（使用字符串方法，避免正则转义问题）
          let displayContent = content;
          const startTag = '<mla_call>';
          const endTag = '</mla_call>';
          let startIdx = displayContent.indexOf(startTag);
          while (startIdx !== -1) {
            const endIdx = displayContent.indexOf(endTag, startIdx);
            if (endIdx !== -1) {
              displayContent = displayContent.substring(0, startIdx) + displayContent.substring(endIdx + endTag.length);
              startIdx = displayContent.indexOf(startTag);
            } else {
              break;
            }
          }
          displayContent = displayContent.trim();
          
          console.log('[updateAssistantMessage] 原始内容长度:', content.length);
          console.log('[updateAssistantMessage] 去掉XML后:', displayContent.length);
          
          // Markdown 渲染
          displayContent = renderMarkdown(displayContent);
          
          console.log('[updateAssistantMessage] Markdown渲染后:', displayContent.length);
          
          contentDiv.innerHTML = displayContent;
        }
      }
      messagesDiv.scrollTop = messagesDiv.scrollHeight;
    }

    function appendMessage(msg) {

        const div = document.createElement('div');

              startIdx = displayContent.indexOf(startTag);
            } else {
              break;
            }
          }

          console.log('[appendMessage] 去掉XML后:', displayContent.substring(0, 100));
          
          displayContent = renderMarkdown(displayContent);
          console.log('[appendMessage] Markdown渲染后:', displayContent.substring(0, 100));

    function updateJSONLEvent(callId, event) {

        eventDiv.style.fontWeight = 'bold';
        eventDiv.style.color = 'var(--vscode-terminal-ansiGreen)';
        eventDiv.style.whiteSpace = 'pre-wrap';
      } else if (event.type === 'end') {
        const duration = event.duration_ms ? '(' + (event.duration_ms / 1000).toFixed(1) + 's)' : '';

        eventDiv.style.fontWeight = 'bold';
        eventDiv.style.whiteSpace = 'pre-wrap';
